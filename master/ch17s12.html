Comparisons

  • Traditional comparisons

    th3index {=, <>, <, >, <=, >=} th3index → boolean

    SELECT th3index '[831c02fffffffff@2001-01-01, 880326b885fffff@2001-01-03]' =
  th3index '[831c02fffffffff@2001-01-01, 880326b885fffff@2001-01-03]';
-- true
SELECT th3index '[831c02fffffffff@2001-01-01, 880326b885fffff@2001-01-03]' <>
  th3index '[831c02fffffffff@2001-01-01, 880326b885fffff@2001-01-03]';
-- false
SELECT th3index '[831c02fffffffff@2001-01-01, 880326b885fffff@2001-01-05]' <
  th3index '[871fa44a8ffffff@2001-01-01, 871fa44a9ffffff@2001-01-03]';
-- true

  • Ever and always comparisons

    {h3index,th3index} {?=, ?<>, %=, %<>} {h3index,th3index}

    SELECT th3index '[831c02fffffffff@2001-01-01, 880326b885fffff@2001-01-05]' ?= 
  880326b885fffff::h3index;
-- true
SELECT th3index '871fa44a8ffffff@2001-01-01' ?<> h3index '871fa44aeffffff';
-- true

  • Temporal comparisons

    {h3index,th3index} {#=, #<>} {h3index,th3index}

    SELECT th3index '871fa44a8ffffff@2001-01-01' #= h3index '871fa44a8ffffff';
-- t@2001-01-01
SELECT th3index '871fa44a8ffffff@2001-01-01' #<> h3index '871fa44a8ffffff';
-- f@2001-01-01
